cmake_minimum_required(VERSION 3.10)
project(zed_capture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp rosbag sensor_msgs std_msgs topic_tools diagnostic_msgs message_generation)
add_message_files(FILES VideoStamp.msg ImuStamp.msg TimeAnchor.msg)
generate_messages(DEPENDENCIES std_msgs)
catkin_package(CATKIN_DEPENDS roscpp rosbag sensor_msgs std_msgs topic_tools diagnostic_msgs message_runtime)
find_package(PkgConfig REQUIRED)
pkg_check_modules(USB REQUIRED libusb-1.0)
find_package(Threads REQUIRED)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../../vendor)
# Private static copies; no system install, CUDA, SDK, OpenCV or global ldconfig.
add_library(capture_hidapi STATIC ${VENDOR}/hidapi/libusb/hid.c)
target_include_directories(capture_hidapi PUBLIC ${VENDOR}/hidapi/hidapi PRIVATE ${USB_INCLUDE_DIRS})
target_link_libraries(capture_hidapi PUBLIC ${USB_LIBRARIES} Threads::Threads)
add_library(capture_oc STATIC ${VENDOR}/zed-open-capture/src/videocapture.cpp ${VENDOR}/zed-open-capture/src/sensorcapture.cpp)
target_include_directories(capture_oc PUBLIC ${VENDOR}/zed-open-capture/include)
target_compile_definitions(capture_oc PUBLIC VIDEO_MOD_AVAILABLE SENSORS_MOD_AVAILABLE)
target_link_libraries(capture_oc PUBLIC capture_hidapi Threads::Threads)
add_executable(zed_capture_node src/capture_node.cpp)
target_include_directories(zed_capture_node PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(zed_capture_node capture_oc ${catkin_LIBRARIES} ${CMAKE_DL_LIBS})
add_dependencies(zed_capture_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
# Optional C4 plugin: only dlopened when explicitly enabled. The default node
# has no new OpenCV/TurboJPEG DT_NEEDED entries and creates no shadow thread.
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
find_library(TURBOJPEG_LIBRARY NAMES turbojpeg REQUIRED)
add_library(capture_jpeg_shadow SHARED src/jpeg_shadow.cpp)
target_include_directories(capture_jpeg_shadow PRIVATE include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(capture_jpeg_shadow ${OpenCV_LIBS} ${TURBOJPEG_LIBRARY} ${catkin_LIBRARIES})
add_dependencies(capture_jpeg_shadow ${${PROJECT_NAME}_EXPORTED_TARGETS})
add_executable(capture_jpeg_shadow_test test/jpeg_shadow_test.cpp)
target_compile_options(capture_jpeg_shadow_test PRIVATE -UNDEBUG)
target_include_directories(capture_jpeg_shadow_test PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(capture_jpeg_shadow_test capture_jpeg_shadow ${catkin_LIBRARIES} Threads::Threads)
add_dependencies(capture_jpeg_shadow_test ${${PROJECT_NAME}_EXPORTED_TARGETS})
add_executable(capture_jpeg_startup_probe test/jpeg_startup_probe.cpp)
target_include_directories(capture_jpeg_startup_probe PRIVATE include ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(capture_jpeg_startup_probe ${OpenCV_LIBS} ${TURBOJPEG_LIBRARY} ${catkin_LIBRARIES} Threads::Threads)
add_dependencies(capture_jpeg_startup_probe ${${PROJECT_NAME}_EXPORTED_TARGETS})
add_executable(capture_core_test test/core_test.cpp)
target_include_directories(capture_core_test PRIVATE include)
target_link_libraries(capture_core_test Threads::Threads)

target_compile_options(capture_core_test PRIVATE -UNDEBUG)
add_executable(capture_input_watchdog_test test/input_watchdog_test.cpp)
target_include_directories(capture_input_watchdog_test PRIVATE include)
target_compile_options(capture_input_watchdog_test PRIVATE -UNDEBUG)
target_link_libraries(capture_input_watchdog_test Threads::Threads)
add_executable(capture_imu_time_segments_test test/imu_time_segments_test.cpp)
target_include_directories(capture_imu_time_segments_test PRIVATE include)
target_compile_options(capture_imu_time_segments_test PRIVATE -UNDEBUG)

# SYNTHETIC syscall/USB completions only; these tests never open hardware.
add_executable(capture_video_io_test test/video_io_test.cpp)
target_compile_options(capture_video_io_test PRIVATE -UNDEBUG)
target_link_libraries(capture_video_io_test capture_oc "-Wl,--wrap=select,--wrap=ioctl")
add_executable(capture_hid_io_test test/hid_io_test.c)
target_include_directories(capture_hid_io_test PRIVATE ${VENDOR}/hidapi/hidapi ${USB_INCLUDE_DIRS})
target_compile_options(capture_hid_io_test PRIVATE -UNDEBUG)
target_link_libraries(capture_hid_io_test ${USB_LIBRARIES} Threads::Threads)

# Independent C5-F failure tests and bounded real-input replay.
add_executable(capture_jpeg_files_test test/jpeg_files_test.cpp)
target_compile_options(capture_jpeg_files_test PRIVATE -UNDEBUG)
target_include_directories(capture_jpeg_files_test PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(capture_jpeg_files_test ${catkin_LIBRARIES} Threads::Threads)
add_dependencies(capture_jpeg_files_test ${${PROJECT_NAME}_EXPORTED_TARGETS})
add_executable(capture_jpeg_files_replay test/jpeg_files_replay.cpp)
target_include_directories(capture_jpeg_files_replay PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(capture_jpeg_files_replay capture_jpeg_shadow ${catkin_LIBRARIES} Threads::Threads)
add_dependencies(capture_jpeg_files_replay ${${PROJECT_NAME}_EXPORTED_TARGETS})

# GNSS-R2: same C5-F bounded queue + ingress/bag-write closure; no hardware/master.

target_compile_options(capture_gnss_writer_test PRIVATE -UNDEBUG)
target_include_directories(capture_gnss_writer_test PRIVATE include ${catkin_INCLUDE_DIRS})
